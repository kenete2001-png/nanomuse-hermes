function frDeepFill(target: Record<string, unknown>, extra: Record<string, unknown>): void {
  for (const [key, value] of Object.entries(extra)) {
    const cur = target[key];
    if (value && typeof value === "object" && !Array.isArray(value)) {
      if (!cur || typeof cur !== "object") target[key] = {};
      frDeepFill(target[key] as Record<string, unknown>, value as Record<string, unknown>);
    } else if (cur === undefined) {
      target[key] = value;
    }
  }
}
